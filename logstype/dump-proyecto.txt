from __future__ import annotations

import pandas as pd

from analisis.estadisticas import resumen_estadistico


def tabla_comparativa(
    df_crudo: pd.DataFrame, df_limpio: pd.DataFrame,
) -> pd.DataFrame:
    est_c = resumen_estadistico(df_crudo)
    est_l = resumen_estadistico(df_limpio)
    if est_c.empty or est_l.empty:
        return pd.DataFrame()

    met = ("media", "mediana", "std", "min", "max", "sesgo", "curtosis")
    filas = []
    for col in est_c.index.intersection(est_l.index):
        for m in met:
            a, b = est_c.loc[col, m], est_l.loc[col, m]
            if pd.isna(a) and pd.isna(b):
                continue
            delta = (b - a) / (abs(a) + 1e-9) * 100
            filas.append({
                "columna": col, "estadistico": m,
                "crudo": a, "limpio": b, "delta_%": delta,
            })
    return pd.DataFrame(filas)


def _indexar(df_crudo: pd.DataFrame, df_limpio: pd.DataFrame):
    k = "ruta_original" if "ruta_original" in df_limpio.columns else "ruta"
    return (df_crudo.drop_duplicates("ruta").set_index("ruta"),
            df_limpio.drop_duplicates(k).set_index(k))


def comparativa_pairwise(
    df_crudo: pd.DataFrame, df_limpio: pd.DataFrame,
) -> pd.DataFrame:
    if "ruta" not in df_crudo.columns or "ruta" not in df_limpio.columns:
        return pd.DataFrame()

    cols_num = [
        c for c in df_crudo.select_dtypes(include="number").columns
        if c in df_limpio.columns
    ]
    if not cols_num:
        return pd.DataFrame()

    crudo_idx, limpio_idx = _indexar(df_crudo, df_limpio)
    comunes = crudo_idx.index.intersection(limpio_idx.index)

    filas: list[dict] = []
    for ruta in comunes:
        for col in cols_num:
            v_c = crudo_idx.at[ruta, col]
            v_l = limpio_idx.at[ruta, col]
            if pd.isna(v_c) and pd.isna(v_l):
                continue
            delta = (v_l - v_c) if pd.notna(v_c) and pd.notna(v_l) else None
            delta_pct = (
                (delta / (abs(v_c) + 1e-9)) * 100
                if delta is not None else None
            )
            if delta is None:
                cambio = "sin dato"
            elif delta == 0:
                cambio = "sin cambio"
            else:
                cambio = "aumentÃ³" if delta > 0 else "disminuyÃ³"
            filas.append({
                "ruta": ruta,
                "columna": col,
                "crudo": v_c,
                "limpio": v_l,
                "delta": delta,
                "delta_%": delta_pct,
                "cambio": cambio,
            })
    return pd.DataFrame(filas)


def comparativa_listwise(
    df_crudo: pd.DataFrame, df_limpio: pd.DataFrame,
) -> pd.DataFrame:
    if "ruta" not in df_crudo.columns or "ruta" not in df_limpio.columns:
        return pd.DataFrame()

    cols_num = [
        c for c in df_crudo.select_dtypes(include="number").columns
        if c in df_limpio.columns
    ]
    if not cols_num:
        return pd.DataFrame()

    crudo_idx, limpio_idx = _indexar(df_crudo, df_limpio)
    crudo_idx, limpio_idx = crudo_idx[cols_num], limpio_idx[cols_num]

    comunes = crudo_idx.index.intersection(limpio_idx.index)
    eliminadas = crudo_idx.index.difference(limpio_idx.index)

    filas: list[dict] = []
    for ruta in comunes:
        a = crudo_idx.loc[ruta].to_numpy(dtype=float)
        b = limpio_idx.loc[ruta].to_numpy(dtype=float)
        mascara = ~(pd.isna(a) | pd.isna(b))
        if not mascara.any():
            distancia = 0.0
        else:
            distancia = float(((a[mascara] - b[mascara]) ** 2).sum() ** 0.5)
        filas.append({
            "ruta": ruta,
            "distancia_l2": distancia,
            "similar": distancia < 1e-6,
            "eliminada": False,
        })
    for ruta in eliminadas:
        filas.append({
            "ruta": ruta,
            "distancia_l2": None,
            "similar": False,
            "eliminada": True,
        })
    return pd.DataFrame(filas)

from __future__ import annotations
import pandas as pd


def _moda_por_columna(num: pd.DataFrame) -> pd.Series:
    resultado: dict[str, float | None] = {}
    for col in num.columns:
        m = num[col].mode(dropna=True)
        resultado[col] = m.iloc[0] if not m.empty else None
    return pd.Series(resultado, dtype="object")


def resumen_estadistico(df: pd.DataFrame) -> pd.DataFrame:
    num = df.select_dtypes(include="number")
    if num.empty:
        return pd.DataFrame()

    media = num.mean()
    std = num.std()
    return pd.DataFrame({
        "media": media,
        "mediana": num.median(),
        "moda": _moda_por_columna(num),
        "std": std,
        "varianza": num.var(),
        "min": num.min(),
        "p25": num.quantile(0.25),
        "p75": num.quantile(0.75),
        "max": num.max(),
        "sesgo": num.skew(),
        "curtosis": num.kurt(),
        "cv": std / (media.abs() + 1e-9),
    })

def _tabla_resumen_numerico(
    df: pd.DataFrame, top_n: int = 8,
) -> pd.DataFrame:
    candidatas = [
        c for c in (
            "brillo_medio", "contraste_rms", "varianza_laplaciano",
            "ruido_estimado", "colorido", "hist_entropia",
            "saturacion_media", "tamano_bytes",
        )
        if c in df.columns and pd.api.types.is_numeric_dtype(df[c])
    ]

    filas: list[dict] = []
    for col in candidatas:
        s = pd.to_numeric(df[col], errors="coerce").dropna()
        if s.empty:
            continue
        q1, q3 = s.quantile(0.25), s.quantile(0.75)
        iqr = q3 - q1
        lo, hi = q1 - 1.5 * iqr, q3 + 1.5 * iqr
        n_out_iqr = int(((s < lo) | (s > hi)).sum())

        mu = s.mean()
        sigma = s.std(ddof=0)
        n_out_z = 0
        if sigma > 0:
            n_out_z = int((((s - mu) / sigma).abs() > 3).sum())

        filas.append({
            "columna": col,
            "n": int(s.size),
            "media": round(float(mu), 4),
            "mediana": round(float(s.median()), 4),
            "std": round(float(sigma), 4),
            "min": round(float(s.min()), 4),
            "p25": round(float(q1), 4),
            "p75": round(float(q3), 4),
            "max": round(float(s.max()), 4),
            "sesgo": round(float(s.skew()), 4),
            "curtosis": round(float(s.kurt()), 4),
            "outliers_iqr": n_out_iqr,
            "outliers_z": n_out_z,
        })
    return pd.DataFrame(filas)


def _tabla_top_bottom(
    df: pd.DataFrame, col: str, etiqueta: str, n: int = 5,
) -> pd.DataFrame:
    if col not in df.columns or df.empty:
        return pd.DataFrame()
    columnas = [c for c in ("ruta", "clase", col) if c in df.columns]
    s = pd.to_numeric(df[col], errors="coerce")
    validos = df.loc[s.notna(), columnas].copy()
    if validos.empty:
        return pd.DataFrame()

    top = validos.nlargest(n, col).assign(ranking=[f"top_{i+1}" for i in range(min(n, len(validos)))])
    bottom = validos.nsmallest(n, col).assign(ranking=[f"bottom_{i+1}" for i in range(min(n, len(validos)))])
    resultado = pd.concat([top, bottom], ignore_index=True)
    resultado.insert(0, "mÃ©trica", etiqueta)
    return resultado

def _tabla_comparativa_resumen(
    comparativa: pd.DataFrame,
) -> pd.DataFrame:
    if comparativa is None or comparativa.empty:
        return pd.DataFrame()
    agrup = comparativa.groupby("columna").agg(
        cambios=("delta_%", lambda x: int((x.abs() > 1e-6).sum())),
        delta_medio_abs=("delta_%", lambda x: round(float(x.abs().mean()), 4)),
        delta_max_abs=("delta_%", lambda x: round(float(x.abs().max()), 4)),
    ).reset_index()
    return agrup.sort_values("delta_medio_abs", ascending=False)

                float(listwise["distancia_l2"].dropna().mean()), 6

            "distancia_max": round(
                float(listwise["distancia_l2"].dropna().max()), 6
            ),
        }])
        secciones.append(SeccionEstadisticas(
            "Listwise_Resumen", tabla_l,
        ))

    return EstadisticasGenerales(secciones)

from __future__ import annotations
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Any

import pandas as pd


@dataclass
class ResultadoReglaEstandar:
    nombre: str
    columna: str
    cumple: pd.Series
    detalle: str = ""
    parametros: dict = field(default_factory=dict)

    @property
    def n_ok(self) -> int:
        return int(self.cumple.sum())

    @property
    def n_total(self) -> int:
        return int(len(self.cumple))


class ReglaEstandar(ABC):
    nombre: str = "base"

    @abstractmethod
    def evaluar(self, df: pd.DataFrame) -> ResultadoReglaEstandar: ...


class ReglaRango(ReglaEstandar):

    def __init__(self, columna: str, minimo: float, maximo: float,
                 nombre: str | None = None):
        self.columna = columna
        self.minimo = minimo
        self.maximo = maximo
        self.nombre = nombre or f"{columna}âˆˆ[{minimo},{maximo}]"

    def evaluar(self, df: pd.DataFrame) -> ResultadoReglaEstandar:
        if self.columna not in df:
            vacio = pd.Series(True, index=df.index)
            return ResultadoReglaEstandar(
                self.nombre, self.columna, vacio,
                detalle=f"omitida (columna ausente: {self.columna})",
                parametros={"min": self.minimo, "max": self.maximo},
            )
        serie = pd.to_numeric(df[self.columna], errors="coerce")
        cumple = serie.between(self.minimo, self.maximo).fillna(False)
        return ResultadoReglaEstandar(
            self.nombre, self.columna, cumple,
            detalle=f"Rango [{self.minimo}, {self.maximo}]",
            parametros={"min": self.minimo, "max": self.maximo},
        )


class ReglaNoNulo(ReglaEstandar):

    def __init__(self, columna: str, nombre: str | None = None):
        self.columna = columna
        self.nombre = nombre or f"{columna} no nulo"

    def evaluar(self, df: pd.DataFrame) -> ResultadoReglaEstandar:
        if self.columna not in df:
            vacio = pd.Series(True, index=df.index)
            return ResultadoReglaEstandar(
                self.nombre, self.columna, vacio,
                detalle=f"omitida (columna ausente: {self.columna})",
            )
        col = df[self.columna]
        cumple = col.notna() & (col.astype(str).str.strip() != "")
        return ResultadoReglaEstandar(
            self.nombre, self.columna, cumple,
            detalle="Valor presente",
        )


class ReglaZScoreMax(ReglaEstandar):

    def __init__(self, columna: str, umbral: float = 3.0,
                 nombre: str | None = None):
        self.columna = columna
        self.umbral = umbral
        self.nombre = nombre or f"{columna} |z|â‰¤{umbral}"

    def evaluar(self, df: pd.DataFrame) -> ResultadoReglaEstandar:
        if self.columna not in df:
            vacio = pd.Series(True, index=df.index)
            return ResultadoReglaEstandar(
                self.nombre, self.columna, vacio,
                detalle=f"omitida (columna ausente: {self.columna})",
            )
        serie = pd.to_numeric(df[self.columna], errors="coerce")
        mu = serie.mean()
        sigma = serie.std(ddof=0)
        if sigma == 0 or pd.isna(sigma):
            cumple = pd.Series(True, index=df.index)
        else:
            z = ((serie - mu) / sigma).abs()
            cumple = (z <= self.umbral).fillna(False)
        return ResultadoReglaEstandar(
            self.nombre, self.columna, cumple,
            detalle=f"Umbral |z| â‰¤ {self.umbral}",
            parametros={"umbral": self.umbral},
        )


class ReglaPersonalizada(ReglaEstandar):

    def __init__(self, nombre: str, columna: str,
                 predicado, detalle: str = ""):
        self.nombre = nombre
        self.columna = columna
        self._predicado = predicado
        self._detalle = detalle

    def evaluar(self, df: pd.DataFrame) -> ResultadoReglaEstandar:

        )


@dataclass
class ResultadoEstandar:
    reglas: list[ResultadoReglaEstandar]
    cumple_global: pd.Series
    aprobadas: pd.DataFrame
    rechazadas: pd.DataFrame
    resumen: dict = field(default_factory=dict)


class EstandarDataset:

    def __init__(self, nombre: str, reglas: list[ReglaEstandar]):
        self.nombre = nombre
        self._reglas = list(reglas)

    @property
    def reglas(self) -> list[ReglaEstandar]:
        return list(self._reglas)

    def agregar_regla(self, regla: ReglaEstandar) -> None:
        self._reglas.append(regla)

    def evaluar(self, df: pd.DataFrame) -> ResultadoEstandar:
        resultados: list[ResultadoReglaEstandar] = []
        cumple_global = pd.Series(True, index=df.index)

        for regla in self._reglas:
            r = regla.evaluar(df)
            resultados.append(r)
            cumple_global &= r.cumple

        aprobadas = df.loc[cumple_global].copy()
        rechazadas = df.loc[~cumple_global].copy()

        resumen = {
            "estandar": self.nombre,
            "total_filas": len(df),
            "aprobadas": int(cumple_global.sum()),
            "rechazadas": int((~cumple_global).sum()),
            "porcentaje_aprobacion": (
                float(cumple_global.mean() * 100) if len(df) else 0.0
            ),
            "reglas": [
                {
                    "nombre": r.nombre,
                    "columna": r.columna,
                    "aprobadas": r.n_ok,
                    "total": r.n_total,
                    "porcentaje": (r.n_ok / r.n_total * 100) if r.n_total else 0.0,
                    "detalle": r.detalle,
                    "parametros": r.parametros,
                }
                for r in resultados
            ],
        }
        return ResultadoEstandar(
            reglas=resultados,
            cumple_global=cumple_global,
            aprobadas=aprobadas,
            rechazadas=rechazadas,
            resumen=resumen,
        )

def _plano(df):
    if df is None:
        return None
    df = df.copy()
    for c in df.columns:
        if df[c].dtype == object and df[c].map(
            lambda x: isinstance(x, (list, tuple, set, dict))
        ).any():
            df[c] = df[c].map(
                lambda x: ", ".join(map(str, x))
                if isinstance(x, (list, tuple, set)) else _celda(x)
            )
    return df


def _resumen_metricas(m: dict) -> pd.DataFrame:
    filas = []
    for k, v in m.items():
        if isinstance(v, dict):
            filas += [{"etapa": k, "mÃ©trica": kk, "valor": _celda(vv)}
                      for kk, vv in v.items()]
        else:
            filas.append({"etapa": "global", "mÃ©trica": k, "valor": _celda(v)})
    return pd.DataFrame(filas)

def _vincular_refs(chart, hoja: Worksheet, datos_df: pd.DataFrame) -> None:
    refs = getattr(chart, "_meta_refs", None)
    if refs is None:
        return
    n = refs["n_filas"]
    filas = (2, n + 1)

    if isinstance(chart, PieChart):
        idx_cat = refs["idx_cat"]
        idx_val = refs["cols_valores"][0]
        cats = Reference(hoja, min_col=idx_cat, min_row=filas[0], max_row=filas[1])
        vals = Reference(hoja, min_col=idx_val, min_row=filas[0], max_row=filas[1])
        chart.add_data(vals, titles_from_data=False)
        chart.set_categories(cats)
        chart.dataLabels = DataLabelList()
        chart.dataLabels.showPercent = True

    elif isinstance(chart, (BarChart, LineChart)):
        idx_cat = refs["idx_cat"]
        cats = Reference(hoja, min_col=idx_cat, min_row=filas[0], max_row=filas[1])
        chart.set_categories(cats)
        for idx_val in refs["cols_valores"]:
            vals = Reference(
                hoja, min_col=idx_val, min_row=filas[0] - 1, max_row=filas[1],
            )
            chart.add_data(vals, titles_from_data=True)

    elif isinstance(chart, ScatterChart):
        idx_x = refs["idx_x"]
        idx_y = refs["idx_y"]
        xs = Reference(hoja, min_col=idx_x, min_row=filas[0], max_row=filas[1])
        ys = Reference(hoja, min_col=idx_y, min_row=filas[0], max_row=filas[1])
        series = Series(ys, xs, title=datos_df.columns[idx_y - 1])
        chart.series.append(series)

    try:
        del chart._meta_refs
    except AttributeError:
        pass


def insertar_grafica(
    writer,
    definicion: DefinicionGrafica,
    datos_df: pd.DataFrame,
) -> None:
    if definicion.tipo not in _CONSTRUCTORES:
        raise ValueError(f"tipo de grÃ¡fica desconocido: {definicion.tipo}")

    hoja = writer.sheets.get(definicion.hoja_datos)
    if hoja is None or datos_df is None or datos_df.empty:
        return
    if definicion.col_categorias not in datos_df.columns:
        return
    if not all(c in datos_df.columns for c in definicion.cols_valores):
        return

    chart = _CONSTRUCTORES[definicion.tipo](datos_df, definicion)
    _vincular_refs(chart, hoja, datos_df)
    chart.width = definicion.ancho
    chart.height = definicion.alto
    hoja.add_chart(chart, definicion.posicion)

from __future__ import annotations
import json
from dataclasses import asdict, is_dataclass
from pathlib import Path
from typing import Any

import numpy as np
import pandas as pd


def _sanear(o: Any) -> Any:
    if isinstance(o, dict):
        return {str(k): _sanear(v) for k, v in o.items()}
    if isinstance(o, (list, tuple, set)):
        return [_sanear(v) for v in o]
    if isinstance(o, (float, np.floating)):
        return None if (np.isnan(o) or np.isinf(o)) else float(o)
    if isinstance(o, np.integer):
        return int(o)
    if isinstance(o, np.bool_):
        return bool(o)
    return o


def _a_json(obj: Any) -> Any:
    if is_dataclass(obj) and not isinstance(obj, type):
        return asdict(obj)
    if isinstance(obj, (np.integer,)):
        return int(obj)
    if isinstance(obj, (np.floating,)):
        return None if np.isnan(obj) else float(obj)
    if isinstance(obj, np.ndarray):
        return obj.tolist()
    if isinstance(obj, pd.Series):
        return obj.to_dict()
    if isinstance(obj, pd.DataFrame):
        return obj.to_dict(orient="records")
    if isinstance(obj, Path):
        return str(obj)
    if isinstance(obj, set):
        return list(obj)
    if isinstance(obj, pd.Timestamp):
        return obj.isoformat()
    if pd.isna(obj):
        return None
    raise TypeError(f"No serializable: {type(obj)} â†’ {obj!r}")

        self._on_progreso = on_progreso

        self._config[clave].parametros = defaults

        return PipelinePreprocesamiento(

            on_log=self._on_log, on_progreso=self._on_progreso,

def _entropia(hist: np.ndarray) -> float:
    p = hist.astype(np.float64) / (hist.sum() + 1e-12)
    p = p[p > 0]
    return float(-(p * np.log2(p)).sum())


def _colorido(img_bgr: np.ndarray) -> float:
    # Hasler & SÃ¼sstrunk (2003)
    b, g, r = cv2.split(img_bgr.astype(np.float32))
    rg = r - g
    yb = 0.5 * (r + g) - b
    std_root = np.sqrt(rg.std() ** 2 + yb.std() ** 2)
    mean_root = np.sqrt(rg.mean() ** 2 + yb.mean() ** 2)
    return float(std_root + 0.3 * mean_root)


def extraer_caracteristicas(ruta: Path) -> CaracteristicasImagen | None:
    img = cv2.imread(str(ruta))
    if img is None:
        return None
    alto, ancho, canales = img.shape
    b, g, r = cv2.split(img)

    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    h, s, v = cv2.split(hsv)

    lab = cv2.cvtColor(img, cv2.COLOR_BGR2LAB)
    l, a, bb = cv2.split(lab)

    gris = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

    # Nitidez
    var_lap = float(cv2.Laplacian(gris, cv2.CV_64F).var())

    # Ruido estimado: diferencia con desenfoque gaussiano
    desenfoque = cv2.GaussianBlur(gris, (5, 5), 0)
    ruido = float(np.std(gris.astype(np.float32) - desenfoque.astype(np.float32)))

    # Histograma de luminancia
    hist = cv2.calcHist([gris], [0], None, [256], [0, 256]).flatten()
    p = hist / (hist.sum() + 1e-12)
    valores = np.arange(256)
    media_h = float((p * valores).sum())
    std_h = float(np.sqrt((p * (valores - media_h) ** 2).sum()))
    sesgo_h = float((p * (valores - media_h) ** 3).sum() / (std_h ** 3 + 1e-12))
    curtosis_h = float((p * (valores - media_h) ** 4).sum() / (std_h ** 4 + 1e-12))

    # Bordes
    canny = cv2.Canny(gris, 100, 200)
    densidad_bordes = float((canny > 0).mean())
    sobel_x = cv2.Sobel(gris, cv2.CV_64F, 1, 0, ksize=3)
    sobel_y = cv2.Sobel(gris, cv2.CV_64F, 0, 1, ksize=3)
    grad_medio = float(np.sqrt(sobel_x**2 + sobel_y**2).mean())

    # Brillo / contraste
    brillo = float(gris.mean())
    contraste = float(gris.std())
    p5, p95 = np.percentile(gris, [5, 95])
    rango_din = float(p95 - p5)

    # Balance de blancos
    r_mean = float(r.mean()) + 1e-9
    g_mean = float(g.mean()) + 1e-9
    b_mean = float(b.mean()) + 1e-9
    wb_rg = r_mean / g_mean
    wb_bg = b_mean / g_mean

    # Color dominante 
    pequena = cv2.resize(img, (64, 64), interpolation=cv2.INTER_AREA)
    cuant = (pequena // 32).reshape(-1, 3)
    codigos = cuant[:, 0] * 64 + cuant[:, 1] * 8 + cuant[:, 2]
    idx = np.bincount(codigos).argmax()
    cb, cg, cr = (idx // 64, (idx // 8) % 8, idx % 8)
    dom_hex = f"#{cr*32:02x}{cg*32:02x}{cb*32:02x}"

    return CaracteristicasImagen(
        ancho=ancho, alto=alto, canales=canales,
        tamano_bytes=ruta.stat().st_size,
        formato=ruta.suffix.lower().lstrip("."),
        r_media=float(r.mean()), g_media=g_mean, b_media=b_mean,
        r_std=float(r.std()), g_std=float(g.std()), b_std=float(b.std()),
        h_media=float(h.mean()), s_media=float(s.mean()), v_media=float(v.mean()),
        l_media=float(l.mean()), a_media=float(a.mean()), b_lab_media=float(bb.mean()),
        brillo_medio=brillo, brillo_std=contraste,
        contraste_rms=contraste, rango_dinamico=rango_din,
        varianza_laplaciano=var_lap, ruido_estimado=ruido,
        hist_entropia=_entropia(hist),
        hist_uniformidad=float((p ** 2).sum()),
        hist_sesgo=sesgo_h, hist_curtosis=curtosis_h,
        densidad_bordes=densidad_bordes, gradiente_medio=grad_medio,
        wb_ratio_rg=wb_rg, wb_ratio_bg=wb_bg,
        colorido=_colorido(img),
        color_dominante_hex=dom_hex,
        saturacion_media=float(s.mean()),
    )

_RE_DIST = re.compile(r"Distancia:\s*(SIN LECTURA|[\d.]+)")
_RE_ESTADO = re.compile(r"ESTADO:\s*([A-Z! ]+)")


class LectorArduinoSerial(FuenteDistancia):

    def __init__(self, puerto: str, baudios: int = 115200,
                 ventana_media: int = 3, verbose: bool = True):
        self._puerto = puerto
        self._baudios = baudios
        self._ventana_media = max(1, ventana_media)
        self._verbose = verbose
        self._hilo: Optional[threading.Thread] = None
        self._parar = threading.Event()
        self._ultima: Optional[LecturaSensor] = None
        self._lock = threading.Lock()
        self._activa = False
        self._ventana: deque[float] = deque(maxlen=self._ventana_media)

    def _log(self, msg: str) -> None:
        if self._verbose:
            print(f"[SENSOR] {msg}")

    def iniciar(self) -> bool:
        self._log(f"iniciar() â†’ puerto={self._puerto} baudios={self._baudios}")
        if serial is None:
            self._log("pyserial NO instalado (pip install pyserial)")
            return False
        self._parar.clear()
        self._ventana.clear()
        self._hilo = threading.Thread(
            target=self._loop, daemon=True, name="lector-arduino"
        )
        self._hilo.start()
        self._log("Esperando 2.5s a que la ESP32 termine de resetear...")
        time.sleep(2.5)
        self._log(f"Estado tras espera: _activa={self._activa}")
        return self._activa

    def detener(self) -> None:
        self._log("detener() â†’ seÃ±alizando hilo...")
        self._parar.set()
        if self._hilo:
            self._hilo.join(timeout=3)
            self._log(f"detener() â†’ hilo vivo={self._hilo.is_alive()}")
        self._activa = False
        self._hilo = None

    def ultima_lectura(self) -> Optional[LecturaSensor]:
        with self._lock:
            return self._ultima

    def esta_activa(self) -> bool:
        return self._activa

    def _loop(self) -> None:

class EtapaDiscretizacion(EtapaTabular):

    tema = "2.3.2"

    @classmethod
    def _parametros_por_defecto(cls) -> dict:
        return {
            "columna": "brillo_medio",
            "bins": 3,
            "estrategia": "ancho",
            "etiquetas": ["oscuro", "medio", "claro"],
        }

    def aplicar(self, df: pd.DataFrame) -> ResultadoEtapa:
        df_antes = df.copy()
        df = df.copy()
        col = self.parametros["columna"]
        if col not in df.columns:
            return ResultadoEtapa(
                nombre=self.nombre, tema=self.tema, df=df, df_antes=df_antes,
                metricas={"filas_afectadas": 0},
                mensaje=f"Columna '{col}' ausente",

        etiquetas = list(self.parametros["etiquetas"])
        if len(etiquetas) != bins:

            if self.parametros["estrategia"] == "frecuencia":
                _, cortes = pd.qcut(
                    df[col], q=bins, retbins=True, duplicates="drop"
                )

                df[nueva] = pd.qcut(
                    df[col], q=bins,
                    labels=etiquetas[:n_reales],
                    duplicates="drop",
                )
            else:
                _, cortes = pd.cut(df[col], bins=bins, retbins=True)
                df[nueva] = pd.cut(df[col], bins=bins, labels=etiquetas)
        except Exception as e:
            return ResultadoEtapa(
                nombre=self.nombre, tema=self.tema, df=df_antes,
                df_antes=df_antes,
                metricas={"filas_afectadas": 0},
                ok=False, error=str(e),
                mensaje=f"DiscretizaciÃ³n fallÃ³: {e}",
            )

        return ResultadoEtapa(
            nombre=self.nombre, tema=self.tema, df=df, df_antes=df_antes,
            metricas={
                "filas_afectadas": int(df[nueva].notna().sum()),
                "cortes": [round(float(c), 4) for c in cortes],
                "conteo": df[nueva].value_counts(dropna=False).to_dict(),
                "columna_creada": nueva,
                "estrategia": self.parametros["estrategia"],
            },
            mensaje=f"{col} â†’ {bins} bins ({self.parametros['estrategia']})",

from __future__ import annotations

from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

import pandas as pd


@dataclass
class ResultadoEtapa:
    nombre: str
    tema: str
    df: pd.DataFrame
    df_antes: pd.DataFrame | None = None
    metricas: dict[str, Any] = field(default_factory=dict)
    mensaje: str = ""
    ok: bool = True
    error: str | None = None

    @property
    def filas_afectadas(self) -> int:
        return int(self.metricas.get("filas_afectadas", 0))


class EtapaPreprocesamiento(ABC):

    nombre: str = "etapa"
    tema: str = "0.0"
    raiz: Path | None = None

    def __init__(self, **parametros: Any) -> None:
        base = self._parametros_por_defecto()
        base.update(parametros or {})
        self.parametros: dict[str, Any] = base

    @classmethod
    def _parametros_por_defecto(cls) -> dict[str, Any]:
        return {}

    @abstractmethod
    def aplicar(self, df: pd.DataFrame) -> ResultadoEtapa: ...

    def descripcion(self) -> str:
        return f"{self.tema} {self.nombre}"

    def reset_parametros(self) -> None:
        self.parametros = self._parametros_por_defecto()

    def _resolver(self, ruta: str) -> Path:
        p = Path(ruta)
        if p.is_absolute() or self.raiz is None:
            return p
        return self.raiz / p

    def _destino(self, df: pd.DataFrame, idx, carpeta: str,
                 ext: str | None = None) -> Path:
        base = self.raiz or Path.cwd()
        rel = Path(str(df.at[idx, "ruta_original"]))
        if rel.is_absolute():
            rel = Path(rel.name)
        destino = base.parent / f"{base.name}_preprocesado" / carpeta / rel
        if ext:
            destino = destino.with_suffix(ext)
        destino.parent.mkdir(parents=True, exist_ok=True)
        return destino

    pass

def _normalizar(s: str, quitar_stop: bool) -> tuple[str, list[str]]:
    s = str(s).lower()

        c for c in unicodedata.normalize("NFKD", s)
        if not unicodedata.combining(c)

    s = re.sub(r"\s+", " ", s).strip()
    tokens = re.findall(r"\w+", s)
    if quitar_stop:
        tokens = [t for t in tokens if t not in _STOPWORDS]
    return " ".join(tokens), tokens

def imputar(df: pd.DataFrame, estrategia: str = "mediana") -> pd.DataFrame:
    salida = df.copy()
    columnas_num = salida.select_dtypes(include="number").columns
    for c in columnas_num:
        if salida[c].isna().sum() == 0:
            continue
        if estrategia == "media":
            salida[c] = salida[c].fillna(salida[c].mean())
        elif estrategia == "moda":
            m = salida[c].mode()
            if not m.empty:
                salida[c] = salida[c].fillna(m.iloc[0])
            salida[c] = salida[c].fillna(salida[c].median())
        else:
            salida[c] = salida[c].fillna(salida[c].median())
    return salida

@dataclass
class ResultadoNormalizacion:
    df: pd.DataFrame
    metodo: str
    columnas_afectadas: list[str]
    parametros: dict = field(default_factory=dict)


def _columnas_numericas(df: pd.DataFrame) -> list[str]:
    return list(df.select_dtypes(include="number").columns)


def normalizar_minmax(df: pd.DataFrame) -> ResultadoNormalizacion:
    salida = df.copy()
    cols = _columnas_numericas(df)
    parametros: dict[str, dict] = {}
    for c in cols:
        s = salida[c]
        if s.notna().sum() < 2:
            continue
        lo, hi = float(s.min()), float(s.max())
        parametros[c] = {"min": lo, "max": hi}
        rango = hi - lo
        salida[c] = 0.0 if rango == 0 else (s - lo) / rango
    return ResultadoNormalizacion(
        df=salida, metodo="minmax",
        columnas_afectadas=cols, parametros=parametros,
    )


def normalizar_zscore(df: pd.DataFrame) -> ResultadoNormalizacion:
    salida = df.copy()
    cols = _columnas_numericas(df)
    parametros: dict[str, dict] = {}
    for c in cols:
        s = salida[c]
        if s.notna().sum() < 2:
            continue
        mu, sigma = float(s.mean()), float(s.std(ddof=0))
        parametros[c] = {"media": mu, "std": sigma}
        salida[c] = 0.0 if sigma == 0 else (s - mu) / sigma
    return ResultadoNormalizacion(
        df=salida, metodo="zscore",
        columnas_afectadas=cols, parametros=parametros,
    )


def normalizar_robusto(df: pd.DataFrame) -> ResultadoNormalizacion:
    salida = df.copy()
    cols = _columnas_numericas(df)
    parametros: dict[str, dict] = {}
    for c in cols:
        s = salida[c]
        if s.notna().sum() < 2:
            continue
        mediana = float(s.median())
        q1, q3 = float(s.quantile(0.25)), float(s.quantile(0.75))
        iqr = q3 - q1
        parametros[c] = {"mediana": mediana, "iqr": iqr}
        salida[c] = 0.0 if iqr == 0 else (s - mediana) / iqr
    return ResultadoNormalizacion(
        df=salida, metodo="robusto",
        columnas_afectadas=cols, parametros=parametros,
    )


def normalizar_l2(df: pd.DataFrame) -> ResultadoNormalizacion:
    salida = df.copy()
    cols = _columnas_numericas(df)
    parametros: dict[str, dict] = {}
    for c in cols:
        s = salida[c]
        norma = float(np.sqrt((s ** 2).sum()))
        parametros[c] = {"norma_l2": norma}
        salida[c] = 0.0 if norma == 0 else s / norma
    return ResultadoNormalizacion(
        df=salida, metodo="l2",
        columnas_afectadas=cols, parametros=parametros,
    )

    "zscore": normalizar_zscore,
    "robusto": normalizar_robusto,
    "l2": normalizar_l2,
}


def metodos_disponibles() -> list[str]:
    return list(_METODOS.keys())


def normalizar(df: pd.DataFrame, metodo: str = "zscore") -> ResultadoNormalizacion:
    if metodo not in _METODOS:
        raise ValueError(
            f"MÃ©todo desconocido: {metodo}. "
            f"Disponibles: {metodos_disponibles()}"
        )
    return _METODOS[metodo](df)

from dataclasses import dataclass
from typing import Iterable
import numpy as np


@dataclass
class ResultadoOutliers:
    indices_zscore: set[int]
    indices_iqr: set[int]
    umbral_z: float
    q1: float
    q3: float
    mediana: float
    media: float
    moda: float | None


def detectar_outliers(valores: Iterable[float], umbral_z: float = 3.0) -> ResultadoOutliers:
    arr = np.array([v for v in valores if v is not None and not np.isnan(v)], dtype=float)
    n = len(arr)
    if n < 5:
        return ResultadoOutliers(set(), set(), umbral_z, 0, 0, 0, 0, None)

    media = float(arr.mean())
    std = float(arr.std(ddof=0))
    mediana = float(np.median(arr))
    q1 = float(np.percentile(arr, 25))
    q3 = float(np.percentile(arr, 75))
    iqr = q3 - q1

    if std > 0:
        z = np.abs((arr - media) / std)
        idx_z = set(np.where(z > umbral_z)[0].tolist())
    else:
        idx_z = set()

    if iqr > 0:
        lo, hi = q1 - 1.5 * iqr, q3 + 1.5 * iqr
        idx_iqr = set(np.where((arr < lo) | (arr > hi))[0].tolist())
    else:
        idx_iqr = set()

    vals, counts = np.unique(arr, return_counts=True)
    moda = float(vals[counts.argmax()]) if len(vals) else None

    return ResultadoOutliers(idx_z, idx_iqr, umbral_z, q1, q3, mediana, media, moda)

from __future__ import annotations

@dataclass
class ConfigEtapa:
    clave: str
    activa: bool = True
    parametros: dict = field(default_factory=dict)


class PipelinePreprocesamiento:

    def __init__(
        self,
        fabrica: dict[str, type[EtapaPreprocesamiento]],
        config: list[ConfigEtapa],
        on_progreso: Callable[[int, int, str], None] | None = None,
        on_log: Callable[[str, str], None] | None = None,
        raiz: Path | None = None,
    ) -> None:
        self.fabrica = fabrica
        self.config = config
        self.on_progreso = on_progreso or (lambda i, n, s: None)
        self.on_log = on_log or (lambda tema, msg: None)
        self.raiz = raiz

    def _instanciar(self, cfg: ConfigEtapa) -> EtapaPreprocesamiento:
        cls = self.fabrica[cfg.clave]
        etapa = cls(**cfg.parametros) if cfg.parametros else cls()
        etapa.raiz = self.raiz
        return etapa

    def ejecutar(self, df: pd.DataFrame) -> list[ResultadoEtapa]:
        activas = [c for c in self.config if c.activa]
        total = len(activas)
        resultados: list[ResultadoEtapa] = []

        for i, cfg in enumerate(activas, start=1):
            etapa = self._instanciar(cfg)
            self.on_progreso(i, total, f"{etapa.tema} {etapa.nombre}")
            try:
                res = etapa.aplicar(df)
            except Exception as e:
                res = ResultadoEtapa(
                    nombre=etapa.nombre, tema=etapa.tema, df=df, ok=False,
                    error=str(e), mensaje=f"Error: {e}",
                )
            resultados.append(res)
            self.on_log(etapa.tema, res.mensaje)
            if res.ok:
                df = res.df
        return resultados

    def ejecutar_solo(self, clave: str, df: pd.DataFrame) -> ResultadoEtapa:
        cfg = next((c for c in self.config if c.clave == clave), None)
        if cfg is None:
            raise KeyError(clave)
        return self._instanciar(cfg).aplicar(df)

class PipelineValidacion:

    def __init__(self, reglas: Iterable[ReglaValidacion]) -> None:
        self._reglas = list(reglas)

    @property
    def reglas(self) -> list[ReglaValidacion]:
        return list(self._reglas)

    ) -> ResultadoRegla:

            severidad=peor.severidad,
            motivo=peor.motivo,
            extras=extras,
        )

from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from pathlib import Path
from typing import Literal

from core.imagen import inspeccionar_imagen, EXTENSIONES_VALIDAS
from core.hashing import sha256_archivo, phash_imagen, distancia_hamming

Severidad = Literal["ok", "revisar", "invalida"]


@dataclass
class ResultadoRegla:
    severidad: Severidad
    motivo: str
    extras: dict = field(default_factory=dict)

    @property
    def valido(self) -> bool:
        return self.severidad != "invalida"


class ReglaValidacion(ABC):
    nombre: str = "base"

    @abstractmethod
    def evaluar(self, archivo: Path, contexto: dict) -> ResultadoRegla: ...


class ReglaExtension(ReglaValidacion):
    nombre = "extension"

    def evaluar(self, archivo, contexto):
        ok = archivo.suffix.lower() in EXTENSIONES_VALIDAS
        if ok:
            return ResultadoRegla("ok", "OK")
        return ResultadoRegla("invalida", "Extension no valida")


class ReglaDecodificacion(ReglaValidacion):
    nombre = "decodificacion"

    def evaluar(self, archivo, contexto):
        info = inspeccionar_imagen(archivo)
        if not info.valida:
            return ResultadoRegla("invalida", info.motivo)
        return ResultadoRegla("ok", "OK", {
            "ancho": info.ancho, "alto": info.alto,
            "formato": info.formato, "modo": info.modo,
        })


class ReglaDuplicadoSHA(ReglaValidacion):
    nombre = "duplicado_sha"

    def evaluar(self, archivo, contexto):
        h = sha256_archivo(archivo)
        vistos = contexto.setdefault("hashes_sha", {})
        if h in vistos:
            return ResultadoRegla("invalida", f"Duplicada SHA de {vistos[h]}", {"hash": h})
        vistos[h] = archivo.name
        return ResultadoRegla("ok", "OK", {"hash": h})


class ReglaDuplicadoPHash(ReglaValidacion):
    nombre = "duplicado_phash"
    UMBRAL = 6

    def evaluar(self, archivo, contexto):
        ph = phash_imagen(archivo)
        if ph is None:
            return ResultadoRegla("ok", "OK: phash omitido (imagehash no instalado)",
                                  {"phash": ""})
        vistos = contexto.setdefault("hashes_phash", [])
        for otro_ph, otro_nombre in vistos:
            if distancia_hamming(ph, otro_ph) <= self.UMBRAL:
                return ResultadoRegla(
                    "invalida", f"Duplicada visual de {otro_nombre}", {"phash": str(ph)}
                )
        vistos.append((ph, archivo.name))
        return ResultadoRegla("ok", "OK", {"phash": str(ph)})


class ReglaCalidadVisual(ReglaValidacion):
    nombre = "calidad_visual"

    def evaluar(self, archivo, contexto):
        import cv2
        from core.calidad import (
            brillo_medio, contraste_rms, varianza_laplaciano,
        )
        img = cv2.imread(str(archivo))
        if img is None:
            return ResultadoRegla("invalida", "No se pudo leer la imagen")
        gris = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        var_lap = varianza_laplaciano(gris)
        brillo = brillo_medio(gris)
        contraste = contraste_rms(gris)
        extras = {
            "varianza_laplaciano": var_lap,
            "brillo_medio": brillo,
            "contraste_rms": contraste,
        }
        if contraste < 5:
            return ResultadoRegla("invalida", "Imagen vacia o casi uniforme", extras)
        if var_lap < 50:
            return ResultadoRegla("revisar", "Revisar: posible borrosidad", extras)
        return ResultadoRegla("ok", "OK", extras)

from __future__ import annotations
import numpy as np
import pandas as pd


def detectar_ruido_caracteristicas(df: pd.DataFrame, k: float = 3.0) -> pd.Series:
    num = df.select_dtypes(include="number").dropna()
    if len(num) < 5:
        return pd.Series(False, index=df.index)

    X = num.to_numpy(dtype=float)
    mu = X.mean(axis=0)
    cov = np.cov(X, rowvar=False) + np.eye(X.shape[1]) * 1e-6
    try:
        inv = np.linalg.inv(cov)
    except np.linalg.LinAlgError:
        return pd.Series(False, index=df.index)

    diff = X - mu
    md = np.sqrt(np.einsum("ij,jk,ik->i", diff, inv, diff))
    umbral = md.mean() + k * md.std()
    es_ruido = pd.Series(False, index=df.index)
    es_ruido.loc[num.index] = md > umbral
    return es_ruido


def suavizar_serie(s: pd.Series, ventana: int = 5) -> pd.Series:
    return s.rolling(window=ventana, center=True, min_periods=1).mean()


def winsorizar(
    df: pd.DataFrame, limites: tuple[float, float] = (0.01, 0.99)
) -> pd.DataFrame:
    salida = df.copy()
    for col in salida.select_dtypes(include="number").columns:
        serie = salida[col]
        if serie.notna().sum() < 2:
            continue
        lo, hi = serie.quantile(limites)
        if lo == hi:
            continue
        salida[col] = serie.clip(lo, hi)
    return salida

@runtime_checkable
class LectorMetadata(Protocol):
    def leer(self, origen: Path) -> list[RegistroMetadata]: ...
    def listar_por_clase(self, origen: Path, clase: str) -> list[RegistroMetadata]: ...
    def existe_hash(self, origen: Path, sha: str) -> RegistroMetadata | None: ...

class EscritorMetadata(Protocol):
    def escribir(self, registros: Iterable[RegistroMetadata], destino: Path) -> Path: ...
    def agregar(self, registro: RegistroMetadata, destino: Path) -> Path: ...


@runtime_checkable

    ...


class RepositorioCSV:

    def leer(self, origen: Path) -> list[RegistroMetadata]:
        origen = Path(origen)
        if not origen.exists():
            return []
        with open(origen, newline="", encoding="utf-8") as f:
            filas = list(csv.DictReader(f))
        return [RegistroMetadata.desde_dict(fila) for fila in filas]

    def escribir(self, registros: Iterable[RegistroMetadata], destino: Path) -> Path:
        destino = Path(destino)
        destino.parent.mkdir(parents=True, exist_ok=True)
        with open(destino, "w", newline="", encoding="utf-8") as f:
            w = csv.DictWriter(f, fieldnames=CAMPOS, extrasaction="ignore")
            w.writeheader()
            for r in registros:
                w.writerow(r.a_dict())
        return destino

    def agregar(self, registro: RegistroMetadata, destino: Path) -> Path:
        registros = self.leer(destino)
        registros.append(registro)
        return self.escribir(registros, destino)

    def listar_por_clase(self, origen: Path, clase: str) -> list[RegistroMetadata]:
        return [r for r in self.leer(origen) if r.clase == clase]

    def existe_hash(self, origen: Path, sha: str) -> RegistroMetadata | None:
        if not sha:
            return None
        for r in self.leer(origen):
            if r.hash == sha and r.es_valido():
                return r
        return None


def nuevo_repositorio() -> RepositorioMetadata:
    return RepositorioCSV()

                and f.stem.startswith(prefijo)
            ):
                resto = f.stem[len(prefijo):]

            str(destino), frame,
            [cv2.IMWRITE_JPEG_QUALITY, CALIDAD_JPEG],
        )
        if not ok:
            print(f"[ERROR] No se pudo guardar {destino}")
            return None

class ClasificadorMobileNet(Clasificador):
    """MobileNetV3 con pesos fine-tuneados a 3 clases.

    pip install torch torchvision
    """

    def __init__(self, ruta_pesos, etiquetas: tuple[str, ...]):
        self._etiquetas = etiquetas
        self._device = torch.device("cpu")

        modelo = models.mobilenet_v3_small(weights=None)
        modelo.classifier[-1] = nn.Linear(modelo.classifier[-1].in_features, len(etiquetas))
        estado = torch.load(str(ruta_pesos), map_location=self._device)
        modelo.load_state_dict(estado)
        modelo.eval()
        self._modelo = modelo.to(self._device)

        self._prep = transforms.Compose([
            transforms.Resize((224, 224)),
            transforms.ToTensor(),
            transforms.Normalize(mean=[0.485, 0.456, 0.406],
                                 std=[0.229, 0.224, 0.225]),
        ])

    def predecir(self, frame_bgr) -> Prediccion | None:
        img = Image.fromarray(frame_bgr[:, :, ::-1])
        tensor = self._prep(img).unsqueeze(0).to(self._device)
        with torch.no_grad():
            logits = self._modelo(tensor)
            probs = torch.softmax(logits, dim=1)[0].cpu().numpy()
        idx = int(np.argmax(probs))
        scores = {c: float(p) for c, p in zip(self._etiquetas, probs)}
        return Prediccion(clase=self._etiquetas[idx],
                          confianza=float(probs[idx]),
                          scores=scores)

    @property
    def clases(self) -> tuple[str, ...]:
        return self._etiquetas

class ServicioClasificacion:
    def __init__(self, clasificador: Clasificador,
                 tamano_buffer: int = 8,
                 umbral_confianza: float = 0.55,
                 frames_consenso: int = 5):
        self._clf = clasificador
        self._buffer = deque(maxlen=tamano_buffer)
        self._umbral = umbral_confianza
        self._frames_consenso = frames_consenso
        self._estado = EstadoClasificacion()

    @property
    def clases(self) -> tuple[str, ...]:
        return self._clf.clases

    @property
    def tiene_modelo(self) -> bool:
        return not isinstance(self._clf, ClasificadorNulo)

    def procesar(self, frame_bgr) -> EstadoClasificacion:
        pred = self._clf.predecir(frame_bgr)
        if pred is None:
            self._buffer.clear()
            self._estado = EstadoClasificacion()
            return self._estado

        self._buffer.append(pred)

        conteo: dict[str, int] = {}
        confianza: dict[str, float] = {}
        for p in self._buffer:
            conteo[p.clase] = conteo.get(p.clase, 0) + 1
            confianza[p.clase] = confianza.get(p.clase, 0.0) + p.confianza

        clase_dom = max(conteo, key=conteo.get)
        n = len(self._buffer)
        ratio = conteo[clase_dom] / n
        conf_prom = confianza[clase_dom] / conteo[clase_dom]

        estable = (ratio >= 0.6 and conf_prom >= self._umbral
                   and conteo[clase_dom] >= self._frames_consenso)
        self._estado = EstadoClasificacion(
            clase=clase_dom,
            confianza=conf_prom,
            estable=estable,
            scores=pred.scores,
        )
        return self._estado

    def reset(self) -> None:
        self._buffer.clear()
        self._estado = EstadoClasificacion()

class EventoTrigger(Enum):
    ACTIVAR = auto()
    CAPTURAR = auto()
    DESACTIVAR = auto()


class FaseTrigger(Enum):
    ESPERANDO = auto()
    DETECTADO = auto()
    CERCA = auto()
    CAPTURADO = auto()


@dataclass(frozen=True)
class ConfigTrigger:
    dist_deteccion_cm: float = 40.0
    dist_cerca_cm: float = 25.0
    dist_captura_cm: float = 10.0
    dist_apagado_cm: float = 50.0
    lecturas_estables: int = 3
    cooldown_s: float = 1.0
    timeout_sin_lectura_s: float = 5.0

    def __post_init__(self):
        if not (self.dist_captura_cm < self.dist_cerca_cm < self.dist_deteccion_cm):
            raise ValueError(
                "Debe cumplirse dist_captura_cm < dist_cerca_cm < dist_deteccion_cm"
            )
        if self.dist_apagado_cm <= self.dist_deteccion_cm:
            raise ValueError(
                "dist_apagado_cm debe ser mayor que dist_deteccion_cm"
            )
        if self.lecturas_estables < 1:
            raise ValueError("lecturas_estables debe ser >= 1")


class EstrategiaTrigger(ABC):
    @abstractmethod
    def procesar(
        self, lectura: Optional[LecturaSensor], ahora: float
    ) -> Optional[EventoTrigger]:
        ...

    @abstractmethod
    def reset(self) -> None:
        ...

    @property
    @abstractmethod
    def fase(self) -> FaseTrigger:
        ...


class TriggerPorDistancia(EstrategiaTrigger):
    def __init__(self, config: Optional[ConfigTrigger] = None):
        self._cfg = config or ConfigTrigger()
        self._fase = FaseTrigger.ESPERANDO
        self._candidata: Optional[FaseTrigger] = None
        self._contador = 0
        self._ultima_foto = 0.0
        self._ultima_lectura_valida = 0.0

    @property
    def fase(self) -> FaseTrigger:
        return self._fase

    def reset(self) -> None:
        self._fase = FaseTrigger.ESPERANDO
        self._candidata = None
        self._contador = 0
        self._ultima_foto = 0.0
        self._ultima_lectura_valida = 0.0

    def procesar(
        self, lectura: Optional[LecturaSensor], ahora: float
    ) -> Optional[EventoTrigger]:
        if lectura is None or not lectura.valida:
            return self._manejar_sin_lectura(ahora)

        self._ultima_lectura_valida = ahora
        distancia = lectura.distancia_cm
        assert distancia is not None

        candidata = self._clasificar(distancia)

        if candidata == self._candidata:
            self._contador += 1
        else:
            self._candidata = candidata
            self._contador = 1
        if self._contador < self._cfg.lecturas_estables:
            return None

        if candidata == self._fase:
            if distancia <= self._cfg.dist_captura_cm:
                return self._decidir_evento(self._fase, candidata, distancia, ahora)
            return None

        anterior, self._fase = self._fase, candidata
        return self._decidir_evento(anterior, candidata, distancia, ahora)

    def _manejar_sin_lectura(self, ahora: float) -> Optional[EventoTrigger]:
        if (
            self._ultima_lectura_valida > 0
            and ahora - self._ultima_lectura_valida >= self._cfg.timeout_sin_lectura_s
            and self._fase != FaseTrigger.ESPERANDO
        ):
            self._fase = FaseTrigger.ESPERANDO
            self._candidata = None
            self._contador = 0
            return EventoTrigger.DESACTIVAR
        return None

    def _clasificar(self, d: float) -> FaseTrigger:
        if d > self._cfg.dist_apagado_cm:
            return FaseTrigger.ESPERANDO
        if d <= self._cfg.dist_captura_cm:
            return FaseTrigger.CERCA
        if d <= self._cfg.dist_cerca_cm:
            return FaseTrigger.CERCA
        return FaseTrigger.DETECTADO

    def _decidir_evento(
        self, anterior: FaseTrigger, nueva: FaseTrigger,
        distancia: float, ahora: float,
    ) -> Optional[EventoTrigger]:
        if nueva == FaseTrigger.ESPERANDO and anterior != FaseTrigger.ESPERANDO:
            return EventoTrigger.DESACTIVAR

        if anterior == FaseTrigger.ESPERANDO and nueva != FaseTrigger.ESPERANDO:
            return EventoTrigger.ACTIVAR

        en_rango_disparo = distancia <= self._cfg.dist_captura_cm
        puede_disparar = ahora - self._ultima_foto >= self._cfg.cooldown_s
        if en_rango_disparo and puede_disparar:
            self._ultima_foto = ahora
            self._fase = FaseTrigger.CAPTURADO
            return EventoTrigger.CAPTURAR

        return None

class ServicioValidacion:
    def __init__(self, pipeline: PipelineValidacion, repo: RepositorioMetadata):
        self._pipeline = pipeline
        self._repo = repo

    def validar_directorio(self, directorio: Path, salida: Path, fuente: str) -> dict:
        directorio = Path(directorio)
        salida = Path(salida)

        if not directorio.exists():
            raise FileNotFoundError(f"No existe el directorio: {directorio}")

        previos = {r.archivo: r for r in self._repo.leer(salida)}
        if previos:
            print(f"[INFO] Metadata previa: {len(previos)} registros (se preservara trazabilidad)")

        registros: list[RegistroMetadata] = []

        # âœ… Estado compartido: UNA sola instancia para todo el run.
        # Las reglas mutan estos objetos in-place y el estado sobrevive
        # aunque un archivo corte el pipeline antes de deduplicar.
        estado: dict = {"hashes_sha": {}, "hashes_phash": []}

        total = validas = invalidas = revisar = 0

        for clase_dir in sorted(directorio.iterdir()):
            if not clase_dir.is_dir() or clase_dir.name not in CLASES_VALIDAS:
                continue
            clase = clase_dir.name
            print(f"Validando clase: {clase}")

            for archivo in sorted(clase_dir.iterdir()):
                if not archivo.is_file():
                    continue
                total += 1

                # ctx comparte referencias con `estado` para los acumuladores
                ctx = {
                    "hashes_sha": estado["hashes_sha"],
                    "hashes_phash": estado["hashes_phash"],
                    "clase": clase,
                    "ruta_rel": ruta_relativa(archivo, BASE_DIR),
                    "fuente": fuente,
                }
                resultado = self._pipeline.validar_archivo(archivo, ctx)

                previo = previos.get(archivo.name)
                reg = RegistroMetadata(
                    archivo=archivo.name,
                    clase=clase,
                    ruta=ctx["ruta_rel"],
                    fuente=(previo.fuente if previo and previo.fuente else fuente),
                    fecha_captura=(previo.fecha_captura if previo else ""),
                    hora_captura=(previo.hora_captura if previo else ""),
                    camara=(previo.camara if previo else ""),
                    sesion=(previo.sesion if previo else ""),
                    version_dataset=(previo.version_dataset if previo else ""),
                    ancho=resultado.extras.get("ancho", ""),
                    alto=resultado.extras.get("alto", ""),
                    formato=resultado.extras.get("formato", ""),
                    tamano_bytes=archivo.stat().st_size,
                    hash=resultado.extras.get("hash", ""),
                    phash=resultado.extras.get("phash", ""),
                    valido=resultado.valido,
                    motivo=resultado.motivo,
                )
                registros.append(reg)

                if resultado.severidad == "invalida":
                    invalidas += 1
                elif resultado.severidad == "revisar":
                    revisar += 1
                    validas += 1
                else:
                    validas += 1

        self._repo.escribir(registros, salida)

        resumen = {
            "directorio": str(directorio),
            "salida": str(salida),
            "total": total,
            "validas": validas,
            "invalidas": invalidas,
            "revisar": revisar,
        }
        self._imprimir_resumen(resumen)
        return resumen

class _WorkerPipeline(QThread):

    progreso = pyqtSignal(str)
    terminado = pyqtSignal(object)   
    error = pyqtSignal(str)

    def __init__(self, raiz: Path, metodo_norm: str, etapas=None, parent=None):
        super().__init__(parent)
        self._raiz = raiz
        self._metodo = metodo_norm
        self._etapas = etapas

    def run(self):
        try:
            pipeline = PipelineAnalisis(metodo_normalizacion=self._metodo)
            resultado = pipeline.analizar(
                self._raiz, al_progresar=self.progreso.emit,
                etapas_config=self._etapas,
            )
            self.terminado.emit(resultado)
        except Exception as e:
            self.error.emit(f"{type(e).__name__}: {e}")

class _Tarea(QThread):
    terminado = pyqtSignal(object)
    error = pyqtSignal(str)

    def __init__(self, fn, parent=None):
        super().__init__(parent)
        self._fn = fn

    def run(self):
        try:
            self.terminado.emit(self._fn())
        except Exception as e:
            self.error.emit(f"{type(e).__name__}: {e}")

    COLOR_TEXTO = QColor("#cdd6f4")     # texto base
    COLOR_NULO = QColor("#6c7086")      # gris para NaN

    def __init__(self, parent=None):
        super().__init__(parent)
        self.setEditTriggers(QAbstractItemView.NoEditTriggers)
        self.setSelectionBehavior(QAbstractItemView.SelectRows)
        self.setSelectionMode(QAbstractItemView.SingleSelection)
        self.setAlternatingRowColors(True)
        self.setShowGrid(False)
        self.verticalHeader().setVisible(False)
        self.horizontalHeader().setSectionResizeMode(
            QHeaderView.ResizeToContents
        )
        self.horizontalHeader().setStretchLastSection(True)
        self.setWordWrap(False)

    def cargar(self, df: pd.DataFrame | None, max_filas: int = 500) -> None:
        self.clear()
        if df is None or df.empty:
            self.setRowCount(0)
            self.setColumnCount(0)
            return

        recorte = df.head(max_filas)
        self.setRowCount(len(recorte))
        self.setColumnCount(len(recorte.columns))
        self.setHorizontalHeaderLabels([str(c) for c in recorte.columns])

        for i, (_, fila) in enumerate(recorte.iterrows()):
            for j, valor in enumerate(fila):
                item = self._crear_item(valor)
                self.setItem(i, j, item)

    def _crear_item(self, valor) -> QTableWidgetItem:
        item = QTableWidgetItem(self._formatear(valor))

        if isinstance(valor, (int, float)) and not isinstance(valor, bool):
            item.setTextAlignment(Qt.AlignRight | Qt.AlignVCenter)
        else:
            item.setTextAlignment(Qt.AlignLeft | Qt.AlignVCenter)

        if pd.isna(valor):
            item.setForeground(self.COLOR_NULO)
        elif isinstance(valor, bool):
            item.setForeground(self.COLOR_TEXTO)
        elif isinstance(valor, int):
            item.setForeground(self.COLOR_ENTEROS)
        elif isinstance(valor, float):
            item.setForeground(self.COLOR_FLOTANTES)
        else:
            item.setForeground(self.COLOR_TEXTO)

        return item

    def _formatear(valor) -> str:
        if pd.isna(valor):
            return "â€”"
        if isinstance(valor, bool):
            return "sÃ­" if valor else "no"
        if isinstance(valor, float):
            if abs(valor) >= 1e4 or (0 < abs(valor) < 1e-3):
                return f"{valor:.3e}"
            return f"{valor:.4f}"
        return str(valor)
